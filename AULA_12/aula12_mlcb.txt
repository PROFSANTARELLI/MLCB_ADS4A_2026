O que é o Streamlit?

Uma biblioteca Python que transforma um script em um aplicativo web interativo, sem HTML, CSS ou JavaScript.
Você escreve código Python normal, e com comandos desenha a tela: título, texto, botão, gráfico, tabela.

Comandos mais usados:
Comando	                                      O que faz
st.title, st.write                            Mostram títulos e textos
st.text_input, st.slider, st.selectbox	      Recebem dados do usuário
st.button	                                    Botão que devolve True no clique
st.line_chart, st.bar_chart                   Gráficos rápidos
st.dataframe	                                Tabela interativa
st.sidebar	                                  Coloca elementos na barra lateral
st.session_state	                            Memória entre as execuções

Como rodar:
pip install streamlit
streamlit run app_streamlit.py (O app abre no navegador, normalmente em http://localhost:8501)

EXEMPLO EM CÓDIGO:

import streamlit as st

st.title(" Aprendendo Streamlit")

# Inicialização da Memória da Sessão
import streamlit as st

# 1. VERIFICAÇÃO E INICIALIZAÇÃO DA MEMÓRIA DE SESSÃO (SESSION STATE) - O Streamlit recarrega o script do zero a cada clique.
# no 'st.session_state', ela seria resetada para 0 a cada interatividade na tela.
if "contador_doacoes" not in st.session_state:

    # Se a chave 'contador_doacoes' ainda não existir na memória da sessão atual do usuário, nós a criamos e a inicializamos com o valor 0.
    st.session_state.contador_doacoes = 0

# 2. LAYOUT EM COLUNAS - Cria 2 colunas lado a lado na interface gráfica para organizar visualmente os botões.
col1, col2 = st.columns(2)

# 3. LÓGICA E BOTÃO DA PRIMEIRA COLUNA (INCREMENTO) - O bloco 'with col1:' indica que tudo dentro dele será desenhado na primeira coluna (esquerda).
with col1:
    # 'st.button()' desenha o botão na tela e retorna True APENAS no momento em que é clicado.
    if st.button("Registrar Doação (+10 kg)"):
        # Se o usuário clicar neste botão, adicionamos +10 ao valor que já está guardado no session_state.
        st.session_state.contador_doacoes += 10

# 4. LÓGICA E BOTÃO DA SEGUNDA COLUNA (RESET) - O bloco 'with col2:' indica que tudo dentro dele será desenhado na segunda coluna (direita).
with col2:
    # Desenha o segundo botão na tela.
    if st.button("Zerar Painel"):
        # Se o usuário clicar neste botão, redefinimos o contador guardado diretamente para 0.
        st.session_state.contador_doacoes = 0

# 5. EXIBIÇÃO DINÂMICA DA MÉTRICA - Desenha um componente visual de cartão de destaque (KPI/Métrica) na interface.
# O parâmetro 'label' é o título do cartão e 'value' é o texto exibido em destaque.
# Usamos uma f-string para formatar dinamicamente o número atualizado do 'session_state' seguido de 'kg'.
st.metric(
    label="Total de Alimentos no Dia", 
    value=f"{st.session_state.contador_doacoes} kg"
)

import streamlit as st
import pandas as pd
import sqlite3
from nlu import MotorNLU
from entities import ExtratorEntidades
from matchmaking import MotorMatchmaking

# Configuração da página do Streamlit
st.set_page_config(
    page_title="MESAFARTAI - Combate à Fome",
    page_icon=" ", # TODO: coloque um ícone aqui
    layout="wide"
)

# Inicializar instâncias dos motores
nlu = MotorNLU()
extrator = ExtratorEntidades()
matchmaker = MotorMatchmaking()

st.title(" ") # TODO: escolha um título aqui
st.caption(" ") # TODO: escolha um subtítulo aqui

# Criação de abas para separar o Chatbot do Dashboard
aba_chat, aba_dashboard = st.tabs(["💬 Chatbot de Doações", "📊 Dashboard de Impacto Social"])

with aba_chat:
    st.subheader("Conversar com o Assistente de Doações")
    
    # Historico da sessão de conversa
    if "mensagens" not in st.session_state:
        st.session_state.mensagens = [
            {"role": "assistant", "content": " "} # TODO: crie um mensagem de iniciar o chat
        ]

    # Exibir mensagens anteriores
    for msg in st.session_state.mensagens:
        st.chat_message(msg["role"]).write(msg["content"])

    # Entrada do Usuário
    if prompt := st.chat_input("Ex: Tenho 20 kg de arroz para doar hoje no centro"):
        st.session_state.mensagens.append({"role": "user", "content": prompt})
        st.chat_message("user").write(prompt)

        # 1. Processar NLU
        intencao, confianca = nlu.predizer_intencao(prompt)

        # 2. Resposta baseada na intenção
        if intencao == "cadastrar_doacao" and confianca >= 0.60:
            entidades = extrator.processar_texto(prompt)
            match = matchmaker.encontrar_ong_mais_proxima(-23.615, -46.785) # Coordenada exemplo
            
            resposta = f"**Doação Registrada com Sucesso!**\n\n" \
                       f"• **Item:** {entidades['alimento']}\n" \
                       f"• **Quantidade:** {entidades['quantidade_kg']} kg\n" \
                       f"• **Validade Estimada:** {entidades['data_validade']}\n\n"
            
            if match:
                resposta += f"**Match Logístico (KNN):** Encaminhado para **{match['ong_nome']}** (~{match['distancia_estimada']} km)."
            
        elif intencao == "solicitar_alimentos":
            resposta = "Entendemos seu pedido. Estamos buscando doações disponíveis próximas a você no banco de dados."
        else:
            resposta = "Desculpe, não entendi perfeitamente sua mensagem. Poderia informar o alimento e a quantidade em kg?"

        st.session_state.mensagens.append({"role": "assistant", "content": resposta})
        st.chat_message("assistant").write(resposta)
